cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD seeed_xiao_rp2350 CACHE STRING "Board")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(nespico_pio_timing_xiao C CXX ASM)
pico_sdk_init()
add_executable(nespico_pio_timing_xiao main.c)
pico_generate_pio_header(nespico_pio_timing_xiao ${CMAKE_CURRENT_LIST_DIR}/revb_seq.pio)
target_link_libraries(nespico_pio_timing_xiao pico_stdlib hardware_pio hardware_dma hardware_clocks)
pico_enable_stdio_usb(nespico_pio_timing_xiao 1)
pico_enable_stdio_uart(nespico_pio_timing_xiao 0)
pico_add_extra_outputs(nespico_pio_timing_xiao)
target_compile_definitions(nespico_pio_timing_xiao PRIVATE USBD_MANUFACTURER="LASAGNA2354" USBD_PRODUCT="LASAGNA2354 pio_timing_xiao")
