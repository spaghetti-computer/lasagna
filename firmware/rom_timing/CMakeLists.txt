cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2 CACHE STRING "Board")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(nespico_rom_timing C CXX ASM)
pico_sdk_init()
option(NESPICO_ROM_TIMING_ARMED "Compile the bus-driving code (interface board required)" OFF)
add_executable(nespico_rom_timing main.c)
pico_generate_pio_header(nespico_rom_timing ${CMAKE_CURRENT_LIST_DIR}/rom_read.pio)
target_compile_definitions(nespico_rom_timing PRIVATE NESPICO_ROM_TIMING_ARMED=$<BOOL:${NESPICO_ROM_TIMING_ARMED}>)
target_link_libraries(nespico_rom_timing pico_stdlib hardware_pio hardware_clocks)
pico_enable_stdio_usb(nespico_rom_timing 1)
pico_enable_stdio_uart(nespico_rom_timing 0)
pico_add_extra_outputs(nespico_rom_timing)
target_compile_definitions(nespico_rom_timing PRIVATE USBD_MANUFACTURER="LASAGNA2354" USBD_PRODUCT="LASAGNA2354 rom_timing")
