cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD seeed_xiao_rp2350 CACHE STRING "Board")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(nespico_latch_bench_xiao C CXX ASM)
pico_sdk_init()
add_executable(latch_bench_xiao main.c)
pico_generate_pio_header(latch_bench_xiao ${CMAKE_CURRENT_LIST_DIR}/bench.pio)
target_link_libraries(latch_bench_xiao pico_stdlib hardware_pio hardware_clocks)
pico_enable_stdio_usb(latch_bench_xiao 1)
pico_enable_stdio_uart(latch_bench_xiao 0)
pico_add_extra_outputs(latch_bench_xiao)
target_compile_definitions(latch_bench_xiao PRIVATE USBD_MANUFACTURER="LASAGNA2354" USBD_PRODUCT="LASAGNA2354 latch_bench_xiao")
